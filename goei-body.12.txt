  function flapStyle(look) {
    const { rx, ry } = look;
    const right = look.corner === "br" || look.corner === "tr";
    const bottom = look.corner === "br" || look.corner === "bl";
    const q = rx * rx + ry * ry;
    const pts3 = [[0.5, 0], [0, 0.5], [ry * ry / q, rx * rx / q]];
    const at = ([u, v]) => [right ? 1 - u : u, bottom ? 1 - v : v];
    const poly = pts3.map((p2) => at(p2).map((n3) => `${(n3 * 100).toFixed(2)}%`).join(" ")).join(", ");
    const dx = (right ? -1 : 1) * ry;
    const dy = (bottom ? -1 : 1) * rx;
    const ang = Math.atan2(dx, -dy) * 180 / Math.PI;
    const len = Math.hypot(dx, dy);
    const ux = dx / len;
    const uy = dy / len;
    const L = Math.abs(2 * rx * ux) + Math.abs(2 * ry * uy);
    const pos = (p2) => {
      const [x, y] = at(p2);
      return 0.5 + ((x - 0.5) * 2 * rx * ux + (y - 0.5) * 2 * ry * uy) / L;
    };
    const t0 = pos(pts3[0]);
    const t1 = pos(pts3[2]);
    const st = (k) => `${((t0 + (t1 - t0) * k) * 100).toFixed(1)}%`;
    return `clip-path:polygon(${poly});background:linear-gradient(${ang.toFixed(1)}deg,#a89a52 ${st(0)},#e9dfa8 ${st(0.1)},#fffbe6 ${st(0.32)},#f4ecc2 ${st(0.7)},#d9cd8f ${st(1)})`;
  }
