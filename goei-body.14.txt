  async function handleJournalSettings(action, v, ctx2, env3) {
    const isLoad = action === "jr-def-yes" && defaultsAsk.pane === "jr" && defaultsAsk.what === "load";
    const def = await defaultsAction("jr", action, v, ctx2, env3.refresh);
    if (def !== null) {
      if (def) rec2.info = def;
      if (isLoad) await redrawChatCards(ctx2.charIndex, ctx2.chatIndex);
      env3.render();
      return true;
    }
    if (action === "jr-set") {
      const i = v.indexOf("=");
      if (i < 0) return true;
      const key = v.slice(0, i);
      let val = v.slice(i + 1);
      if (key === "chat_card") val = val === "off" || val === "all" ? val : "mine";
      else return true;
      await setVars4(ctx2, env3, { [JR(key)]: val });
      if (key === "chat_card") await redrawChatCards(ctx2.charIndex, ctx2.chatIndex);
      return true;
    }
    if (action === "jr-step") {
      const key = v.split(/[:=]/)[0];
      const s8 = journalSettings(ctx2.chat.scriptstate);
      const to = (cur, r, hi = r.max) => stepNext(v, cur, r.min, hi);
      let next;
      if (key === "journal_cap") next = to(s8.journalCap, JR_JOURNAL_CAP);
      else if (key === "report_cap") next = to(s8.reportCap, JR_REPORT_CAP, Math.min(JR_REPORT_CAP.max, s8.journalCap));
      else if (key === "list_page") next = to(s8.listPage, JR_LIST_PAGE);
      else if (key === "trail_max") next = to(s8.trailMax, JR_TRAIL_MAX);
      else if (key === "people_max") next = to(s8.peopleMax, JR_PEOPLE_MAX);
      else if (key === "trades_max") next = to(s8.tradesMax, JR_TRADES_MAX);
      else return true;
      if (next === null || String(next) === jrValue(ctx2.chat.scriptstate, key)) return true;
      await setVars4(ctx2, env3, { [JR(key)]: String(next), ...key === "journal_cap" && next < s8.reportCap ? { [JR("report_cap")]: String(next) } : {} });
      return true;
    }
    return false;
  }
